cmake_minimum_required(VERSION 3.20)
project(horncad_mfem_acoustics LANGUAGES CXX)

find_package(MFEM CONFIG REQUIRED)
add_executable(horncad_mfem_interior interior_acoustics.cpp)
target_compile_features(horncad_mfem_interior PRIVATE cxx_std_17)
target_link_libraries(horncad_mfem_interior PRIVATE mfem)

if(MFEM_USE_MPI AND MFEM_USE_SUPERLU)
  find_package(MPI REQUIRED COMPONENTS CXX)
  add_executable(horncad_mfem_interior_parallel parallel_interior_acoustics.cpp)
  target_compile_features(horncad_mfem_interior_parallel PRIVATE cxx_std_17)
  target_link_libraries(horncad_mfem_interior_parallel PRIVATE mfem MPI::MPI_CXX)
else()
  message(STATUS "Parallel FEM backend disabled: MFEM needs MPI and SuperLU_DIST")
endif()
